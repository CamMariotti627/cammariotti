function decimalToBinary(decNumber) {
  var remStack = new Stack(), binaryString = "";
  while (decNumber > 0) {
    remStack.push(decNumber % 2);
    decNumber = Math.floor(decNumber / 2);
  }
  while (!remStack.isEmpty()) {
    binaryString += remStack.pop().toString();
  }
  return binaryString;
}

Priority queue with ordered insertion

PriorityQueue.prototype.enqueue = function (element, priority) {
  var node = new QueueElement(element, priority);
  var added = false;
  for (var i = 0; i < this.items.length; i++) {
    if (node.priority < this.items[i].priority) {
      this.items.splice(i, 0, node);
      added = true;
      break;
    }
  }
  if (!added) this.items.push(node);
};

Maintains sort order on insert rather than sorting after the fact, keeping dequeue O(1).

Linked list with arbitrary-position insert/remove

LinkedList.prototype.insert = function (index, element) {
  var node = new Node(element), current = this.head, previous, i = 0;
  if (index < 0 || index > this.length) return false;
  if (index === 0) {
    node.next = current;
    this.head = node;
  } else {
    while (i++ < index) { previous = current; current = current.next; }
    node.next = current;
    previous.next = node;
  }
  this.length++;
};

Handles the head-insertion edge case separately from mid-list insertion — a common interview-style pointer manipulation.

Set algebra

Set.prototype.union = function (otherSet) {
  var result = new Set();
  this.values().forEach(function (v) { result.add(v); });
  otherSet.values().forEach(function (v) { result.add(v); });
  return result;
};
Set.prototype.intersection = function (otherSet) {
  var result = new Set();
  this.values().forEach(function (v) {
    if (otherSet.has(v)) result.add(v);
  });
  return result;
};

Implements union, intersection, difference, and subset from scratch rather than relying on a built-in Set.

Shortest path (Dijkstra's algorithm)

function dijkstra(graph, src) {
  var dist = graph.map(function () { return Infinity; });
  var visited = graph.map(function () { return false; });
  dist[src] = 0;
  for (var count = 0; count < graph.length; count++) {
    var u = minDistance(dist, visited);
    visited[u] = true;
    for (var v = 0; v < graph.length; v++) {
      if (!visited[v] && graph[u][v] && dist[u] + graph[u][v] < dist[v]) {
        dist[v] = dist[u] + graph[u][v];
      }
    }
  }
  return dist;
}

Computes shortest distances from a source vertex across a weighted adjacency matrix.

Quick sort via partitioning

function quickSort(arr) {
  if (arr.length <= 1) return arr;
  var pivot = arr[Math.floor(arr.length / 2)];
  var left = arr.filter(function (x) { return x < pivot; });
  var middle = arr.filter(function (x) { return x === pivot; });
  var right = arr.filter(function (x) { return x > pivot; });
  return quickSort(left).concat(middle, quickSort(right));
}

Recursive divide-and-conquer sort, paired with a Fisher-Yates shuffle to generate unsorted test data.

Binary search

function binarySearch(sortedArr, target) {
  var low = 0, high = sortedArr.length - 1;
  while (low <= high) {
    var mid = Math.floor((low + high) / 2);
    if (sortedArr[mid] === target) return mid;
    if (sortedArr[mid] < target) low = mid + 1;
    else high = mid - 1;
  }
  return -1;
}

Applied to a real text corpus — sorting and binary-searching the words of a document to check for the presence of specific terms.

Dynamic programming: minimum coin change

function minCoinChangeDP(coins, amount) {
  var dp = [0], used = [];
  for (var i = 1; i <= amount; i++) {
    dp[i] = Infinity;
    for (var j = 0; j < coins.length; j++) {
      if (coins[j] <= i && dp[i - coins[j]] + 1 < dp[i]) {
        dp[i] = dp[i - coins[j]] + 1;
        used[i] = coins[j];
      }
    }
  }
  return dp[amount];
}

Builds a bottom-up DP table and compares the result directly against a greedy solution to the same problem, showing when greedy fails to be optimal.

Recursion with memoization

function longestCollatzChain(limit) {
  var cache = { 1: 1 };
  function chainLength(n) {
    if (cache[n]) return cache[n];
    cache[n] = 1 + chainLength(n % 2 === 0 ? n / 2 : 3 * n + 1);
    return cache[n];
  }
  var best = 1, bestStart = 1;
  for (var s = 2; s < limit; s++) {
    if (chainLength(s) > best) { best = chainLength(s); bestStart = s; }
  }
  return { start: bestStart, length: best };
}

Caches recursive results to make an otherwise expensive brute-force search (Project Euler #14) tractable.

Complexity analysis in practice

// O(n²): compares every pair
function hasDuplicates(arr) {
  for (var i = 0; i < arr.length; i++)
    for (var j = 0; j < arr.length; j++)
      if (i !== j && arr[i] === arr[j]) return true;
  return false;
}

// O(n): single pass using a Set for O(1) lookups
function hasDuplicatesFast(arr) {
  var seen = new Set();
  for (var i = 0; i < arr.length; i++) {
    if (seen.has(arr[i])) return true;
    seen.add(arr[i]);
  }
  return false;
}

Identifies a quadratic bottleneck and rewrites it as a linear-time solution — demonstrating Big O analysis applied to real optimization, not just theory.

Greedy algorithm design (Travelling Salesman)

function greedyTSP(graph, start) {
  var visited = [start], current = start, total = 0;
  while (visited.length < graph.length) {
    var nearest = findNearestUnvisited(graph, current, visited);
    total += graph[current][nearest.index];
    visited.push(nearest.index);
    current = nearest.index;
  }
  total += graph[current][start]; // return to origin
  return { path: visited, total: total };
}

Nearest-neighbor heuristic for a classic NP-hard problem, with an explicit discussion of why the greedy result isn't guaranteed optimal.
